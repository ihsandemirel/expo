import React, { useEffect, useState } from 'react';
import { Text, View, Button, ScrollView, StyleSheet } from 'react-native';
import * as Location from 'expo-location';
import * as Notifications from 'expo-notifications';
import { PrayerTimes, CalculationMethod, Coordinates, Madhab } from 'adhan';

export default function App() {
  const [prayers, setPrayers] = useState({});
  const [marked, setMarked] = useState({});

  useEffect(() => {
    (async () => {
      const { status: locStatus } = await Location.requestForegroundPermissionsAsync();
      const { status: notifStatus } = await Notifications.requestPermissionsAsync();
      if (locStatus !== 'granted' || notifStatus !== 'granted') {
        console.warn('Konum veya bildirim izni verilmedi.');
        return;
      }
      await calculateAndSchedule();
    })();
  }, []);

  const calculateAndSchedule = async () => {
    const { coords } = await Location.getCurrentPositionAsync();
    const params = CalculationMethod.Turkey();
    params.madhab = Madhab.Hanafi;

    const times = new PrayerTimes(
      new Coordinates(coords.latitude, coords.longitude),
      new Date(),
      params
    );

    const allTimes = {
      Fajr: times.fajr,
      Dhuhr: times.dhuhr,
      Asr: times.asr,
      Maghrib: times.maghrib,
      Isha: times.isha
    };
    setPrayers(allTimes);

    // Önceki bildirimleri sil
    const pending = await Notifications.getAllScheduledNotificationsAsync();
    await Promise.all(pending.map(n => Notifications.cancelScheduledNotificationAsync(n.identifier)));

    // Yeni bildirimleri ekle
    Object.entries(allTimes).forEach(([name, date]) => {
      Notifications.scheduleNotificationAsync({
        content: {
          title: `${name} vakti geldi`,
          body: 'Namazını kıldın mı? İşaretlemezsen tekrar hatırlatma yapacağız.',
          sound: true
        },
        trigger: { date }
      });
    });
  };

  const markPrayer = async (name) => {
    setMarked(m => ({ ...m, [name]: true }));
    const pending = await Notifications.getAllScheduledNotificationsAsync();
    for (let n of pending) {
      if (n.content.title.startsWith(name)) {
        await Notifications.cancelScheduledNotificationAsync(n.identifier);
      }
    }
  };

  return (
    <ScrollView contentContainerStyle={styles.container}>
      {Object.entries(prayers).map(([name, date]) => (
        <View key={name} style={styles.row}>
          <Text style={styles.text}>
            {name}: {date.toLocaleTimeString('tr-TR')}
          </Text>
          <Button
            title={marked[name] ? "✓ İşaretlendi" : "Kıldım"}
            onPress={() => markPrayer(name)}
          />
        </View>
      ))}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: { padding: 20 },
  row: { flexDirection: 'row', alignItems: 'center', marginBottom: 16 },
  text: { flex: 1, fontSize: 16 }
});
